# ТестАПІ.ert

1. В глобальному модулі мають бути всі зміни, описані в `global_module_for_1C.md`

2. Створюємо завнішню процедуру `reports\ТестАПІ.ert`

3. На вкладці `Диалог` створюємо три нових поля:
	- ВыбЧисло - тип даних `Число`
	- ВыбДата - тип даних `Дата`
	- ВыбСтрока - тип даних `Строка`

4. Текст, який знаходиться нижче, копіюємо у вкладку  `Модуль`

5. Зберігаємо і запускаємо в 1С.


```text
//====================================================
Функция Сформировать()
	
	data = СоздатьОбъект("СписокЗначений");   
	// дії з числами
	data.ДобавитьЗначение(ВыбЧисло * ВыбЧисло, "squared_number");
	
	// дії з датами (Формат "ДДДММГГГГ"    !!!)
	data.ДобавитьЗначение(Формат(НачМесяца(ВыбДата), "ДДДММГГГГ"), "date_start");
	data.ДобавитьЗначение(Формат(КонМесяца(ВыбДата), "ДДДММГГГГ"), "date_finish");
	
	// дії з рядками (СокрЛП !!!)
	data.ДобавитьЗначение(СокрЛП(Врег(ВыбСтрока)), "uper_name"); 
	
	// дії з масивом (Ключ = Строка(Номер))
	Список1 = СоздатьОбъект("СписокЗначений");  
	Для Номер = 1 По ВыбЧисло Цикл
		Список1.ДобавитьЗначение(Номер, Строка(Номер));
	КонецЦикла;
	data.ДобавитьЗначение(Список1, "array");
	
	// дії з словником
	Список2 = СоздатьОбъект("СписокЗначений");  
	Для Номер = 1 По ВыбЧисло Цикл
		Список2.ДобавитьЗначение(СокрЛП(ВыбСтрока) + " - " + Строка(Номер), "key_" + Строка(Номер));
	КонецЦикла;
	data.ДобавитьЗначение(Список2, "object");
	
	Output = СоздатьОбъект("СписокЗначений");
	Output.ДобавитьЗначение("ok", "status");
	Output.ДобавитьЗначение("test passed", "message");
	Output.ДобавитьЗначение(data, "data");
	
	// перегляд json при інтерактивному запуску
	Если ПустоеЗначение(Форма.Параметр) = 1 Тогда
		Просмотр = СоздатьОбъект("Текст");
		Просмотр.ДобавитьСтроку(list_to_json(Output));
		Просмотр.Показать("json");
	КонецЕсли;
	
	Возврат Output
КонецФункции

//====================================================
Процедура ПриОткрытии()
	Если ПустоеЗначение(Форма.Параметр) = 0 Тогда

		// Отримуємо список значень, переданий з процедури глобального модуля api
		Данные = Форма.Параметр;       
		Input = Данные.ПолучитьЗначение(1, "Input");
		Параметри = Input.Получить("parameters");    
		
		// тестовий приклад параметрів
		//    "parameters": {
		//        "number": 3,
		//        "date": "18.05.2026",
		//        "name": "Текст \"Назва іїє\"",
		//    }
		
		ВыбЧисло = Число(Параметри.Получить("number"));
		ВыбДата = Дата(Параметри.Получить("date"));
		ВыбСтрока = Строка(Параметри.Получить("name"));
		
		Output = Сформировать();		
		
		// Записуємо результат звіту назад у список (другим значенням)
		Данные.УстановитьЗначение(2, Output, "Output");
		Форма.Закрыть();
		
	КонецЕсли;
КонецПроцедуры
```
