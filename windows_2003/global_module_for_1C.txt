	parse_json = СоздатьОбъект("MSScriptControl.ScriptControl");

	код = "              
	|                    
	| function parseJSON(strJSON) {
	|  var tmpFunc = (new Function('return('+strJSON+');'))();
	|  return(tmpFunc);
	| }
	|
	| // Получить элемент массива
	| function aGet(Array, index) {
	|  return(Array[index]);
	| }
	|
	| // Получить ключ пары по индексу
	| function oKey(Obj, index) {
	|  var size = 0, key;
	|  for (key in Obj) {
	|   if (size == index) break;
	|   if (Obj.hasOwnProperty(key)) size++;
	|  }
	|  return(key);
	| }
	|
	| // Получить значение пары по ключу
	| function oValueByKey(Obj, key) {
	|  return(Obj[key]);
	| }
	|
	| //Получить количество пар в объекте
	| Object.size = function(obj) {
	|  var size = 0, key;
	|  for (key in obj) {
	|   if (obj.hasOwnProperty(key)) size++;
	|  }
	|  return(size);
	| }
	|
	| //Получить размер объекта (количество пар в нём)

				сп.ДобавитьЗначение(json_to_list(Значение), Ключ);
			КонецЕсли;

	ИначеЕсли ТипОбъекта = "array" Тогда

			Значение = parse_json.run("aGet", obj, н);

// Основна функція перетворення
Функция list_to_json(Список, Отступ=2) Экспорт
	Пробелы = "                        ";
    Рез = "";
    Кількість = Список.РазмерСписка();
    
    // Визначаємо, чи це масив (ключі — числа) чи об'єкт
    // В 1С 7.7 ми зазвичай використовуємо представлення як ключі
    ПершийКлюч = "";
    Список.ПолучитьЗначение(1, ПершийКлюч);
    
    // Якщо представлення першого елемента можна перетворити в число 1 — вважаємо масивом
    ЦеМасив = 0;
    Если СокрЛП(ПершийКлюч) = "1" Тогда ЦеМасив = 1; КонецЕсли;

    Если ЦеМасив = 1 Тогда 
		Рез = Рез + "[" + РазделительСтрок;
	Иначе 
		Рез = Рез + Лев(Пробелы, Отступ - 2) + "{" + РазделительСтрок;
	КонецЕсли;

    Для н = 1 По Кількість Цикл
        Ключ = "";
        Значение = Список.ПолучитьЗначение(н, Ключ);
        ТипЗ = ТипЗначенияСтр(Значение);
        
        // Формуємо ключ (якщо це об'єкт)
        Если ЦеМасив = 0 Тогда
            Рез = Рез + Лев(Пробелы, Отступ) + """" + json_str_wrapper(Ключ) + """: ";
        КонецЕсли;
        
        // Обробка значення
        Если ТипЗ = "СписокЗначений" Тогда
            Рез = Рез + list_to_json(Значение, Отступ + 2);
        ИначеЕсли ТипЗ = "Число" Тогда
            Рез = Рез + Строка(Значение);
        Иначе
            Рез = Рез + """" + json_str_wrapper(Строка(Значение)) + """";
        КонецЕсли;
        
        Если н < Кількість Тогда 
			Рез = Рез + "," + РазделительСтрок;
		КонецЕсли;
    КонецЦикла;

    Если ЦеМасив = 1 Тогда 
		Рез = Рез + РазделительСтрок + Лев(Пробелы, Отступ - 2) + "]";
	Иначе 
		Рез = Рез + РазделительСтрок + Лев(Пробелы, Отступ - 2) + "}";
	КонецЕсли;
    
    Возврат Рез;

// Головна функція API
// ШляхДоВхідногоФайла - повний шлях до JSON, який створив Python
// Повертає: повний шлях до файлу з результатом (JSON)

Функция api_global(ШляхДоВхідногоФайла) Экспорт      
	Перем ЧЧ, ММ, СС;
	
	// 1. Підготовка контейнера для результату (Output)
	Output = СоздатьОбъект("СписокЗначений");
	ШляхВихідногоФайла = СтрЗаменить(ШляхДоВхідногоФайла, ".json", "_out.json");
	
	// 2. Ініціалізація JScript для парсингу (якщо ще не ініціалізовано)
	Если ПустоеЗначение(parse_json) = 1 Тогда
		init_parse_json(parse_json);
	КонецЕсли;
	
	// 3. Зчитування вхідного файлу
	ТекстВхід = СоздатьОбъект("Текст");
	Попытка
		ТекстВхід.Открыть(ШляхДоВхідногоФайла);
		РядокJSON = "";
		Для н = 1 По ТекстВхід.КоличествоСтрок() Цикл
			РядокJSON = РядокJSON + ТекстВхід.ПолучитьСтроку(н);
		КонецЦикла;
	Исключение
		Output.ДобавитьЗначение("error", "status");
		Output.ДобавитьЗначение("Не вдалося прочитати вхідний файл: " + ОписаниеОшибки(), "message");
		Перейти ~ЗаписРезультату;
	КонецПопытки;
	
	// 4. Конвертація JSON у СписокЗначений (Input)
	Попытка
		jsObj = parse_json.run("parseJSON", РядокJSON);
		Input = json_to_list(jsObj);
	Исключение
		Output.ДобавитьЗначение("error", "status");
		Output.ДобавитьЗначение("Помилка парсингу JSON: " + ОписаниеОшибки(), "message");
		Перейти ~ЗаписРезультату;
	КонецПопытки;
	
	// 5. Пошук команди
	Команда = Input.Получить("procedure");
	Если ПустаяСтрока(Команда) = 1 Тогда
		Output.ДобавитьЗначение("error", "status");
		Output.ДобавитьЗначение("В JSON не знайдено поле 'procedure'", "message");
		Перейти ~ЗаписРезультату;
	КонецЕсли;
	
	// 6. Запуск зовнішньої процедури
	ШляхДоЕрт = КаталогИБ() + "api\reports\" + СокрЛП(Команда) + ".ert";

	Если ФС.СуществуетФайл(ШляхДоЕрт) = 0 Тогда
		Output.ДобавитьЗначение("error", "status");
		Output.ДобавитьЗначение("Файл обробки не знайдено: " + ШляхДоЕрт, "message");
		Перейти ~ЗаписРезультату;
	КонецЕсли;
	
	// Створюємо структуру для обміну зі звітом
	ДаніОбміну = СоздатьОбъект("СписокЗначений");
	ДаніОбміну.ДобавитьЗначение(Input, "Input");
	ДаніОбміну.ДобавитьЗначение("", "Output"); // Сюди звіт запише результат
	
	ЗаписьЖурналаРегистрации(ШляхДоЕрт, "Пакетный режим", "Начато выполнение пакетного задания");

	ТекущееВремя(ЧЧ, ММ, СС);
	СтартВремя = ЧЧ * 3600 + ММ * 60 + СС;

	ОткрытьФормуМодально("Отчет", ДаніОбміну, ШляхДоЕрт);

	ТекущееВремя(ЧЧ, ММ, СС);
	ВремяСекунд = ЧЧ * 3600 + ММ * 60 + СС - СтартВремя;

	
	// Отримуємо відповідь від ERT
	РезультатЗвіту = ДаніОбміну.Получить("Output");
	                                                            
	
	Если ТипЗначенияСтр(РезультатЗвіту) = "СписокЗначений" Тогда
		Output = РезультатЗвіту;
	Иначе
		Output.ДобавитьЗначение("error", "status");
		Output.ДобавитьЗначение("Критична помилка виконання обробки: " + ШляхДоЕрт, "message");
	КонецЕсли;       
	// фіксуємо час виконання обробки
	Output.ДобавитьЗначение(ВремяСекунд, "seconds");

	ЗаписьЖурналаРегистрации(Output.Получить("message"), "Пакетный режим", "Завершено выполнение пакетного задания");

	
	// 7. Формування вихідного файлу
~ЗаписРезультату:
	ТекстВихід = СоздатьОбъект("Текст");
	ТекстВихід.ДобавитьСтроку(list_to_json(Output));
	ТекстВихід.Записать(ШляхВихідногоФайла);
	
	Возврат ШляхВихідногоФайла;
	
КонецФункции
